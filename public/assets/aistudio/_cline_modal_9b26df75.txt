/**
 * URL guard shared by the Studio bridge and the generated extension runtime.
 * Only stable, publicly reachable HTTPS URLs (plus SVG-safe data:image) may
 * cross into the extension â€” blob: URLs die with the tab that created them
 * and must never be persisted into chrome.storage.local.
 */
export function isBridgeableWallpaperUrl(url: unknown): boolean {
  if (typeof url !== 'string') return false;
  const clean = url.trim();
  if (!clean) return false;
  if (/^blob:/i.test(clean)) return false;
  return /^(https?:\/\/|data:image\/)/i.test(clean);
}

function getSafeOrigin(): string {
  if (typeof window !== 'undefined' && window.location?.origin && window.location.origin !== 'null') {
    return window.location.origin;
  }
  return '*';
}
