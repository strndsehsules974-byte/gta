describe('renderExtensionIconPng', () => {
  it('emits valid PNG signature and IHDR dimensions', () => {
    const png = renderExtensionIconPng(128, '#06b6d4');
    expect(Array.from(png.subarray(0, 4))).toEqual(PNG_MAGIC);
    const width = ((png[16] << 24) | (png[17] << 16) | (png[18] << 8) | png[19]) >>> 0;
    const height = ((png[20] << 24) | (png[21] << 16) | (png[22] << 8) | png[23]) >>> 0;
    expect(width).toBe(128);
    expect(height).toBe(128);
  });

  it('clamps requested sizes into the supported 16â€“256 range', () => {
    const tiny = renderExtensionIconPng(1, '#ffffff');
    const width = ((tiny[16] << 24) | (tiny[17] << 16) | (tiny[18] << 8) | tiny[19]) >>> 0;
    expect(width).toBe(16);
  });

  it('tolerates malformed hex colors by falling back to cyan', () => {
    const png = renderExtensionIconPng(16, 'not-a-color');
    expect(Array.from(png.subarray(0, 4))).toEqual(PNG_MAGIC);
  });
});
